    ```tsx

    import { SinPermiso } from "@/components/ui/SinPermiso";
    import { ActivityIndicator, View } from "react-native";

    export default function HistorialLayout() { // (or ResumenLayout)
      const { rol, ready } = useSessionStore((s) => ({ rol: s.rol, ready: s.ready }));

      // Wait for session hydration before enforcing role — prevents flash of restricted content
      if (!ready) {
        return (
          <View className="flex-1 items-center justify-center bg-[#FAFAF9] dark:bg-[#0C0F14]">
            <ActivityIndicator size="large" color="#ea580c" />
          </View>
        );
      }

      if (rol === "CAJERO") {
        return <SinPermiso />;
      }

      return (
        // ... existing Stack or content, preserved exactly ...
      );
    }
    ```
